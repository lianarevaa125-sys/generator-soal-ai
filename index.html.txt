    <script>
        const form = document.getElementById('generatorForm');
        let currentData = null; 
        
        let pdfBase64Data = null;
        let pdfMimeType = null;

        // Logic membaca file PDF ke sistem
        function updateFileName(input) {
            const display = document.getElementById('fileNameDisplay');
            if (input.files && input.files.length > 0) {
                document.getElementById('fileNameText').textContent = input.files[0].name;
                display.classList.remove('hidden');

                const file = input.files[0];
                pdfMimeType = file.type;
                const reader = new FileReader();
                reader.readAsDataURL(file);
                reader.onload = () => {
                    let encoded = reader.result.toString().replace(/^data:(.*,)?/, '');
                    if ((encoded.length % 4) > 0) encoded += '='.repeat(4 - (encoded.length % 4));
                    pdfBase64Data = encoded;
                };
            } else {
                display.classList.add('hidden');
                pdfBase64Data = null;
                pdfMimeType = null;
            }
        }

        // Tab Switcher
        function switchTab(tabName) {
            document.getElementById('tabSoal').classList.remove('active');
            document.getElementById('tabKisi').classList.remove('active');
            document.getElementById('viewSoal').classList.add('hidden');
            document.getElementById('viewKisi').classList.add('hidden');

            if(tabName === 'soal') {
                document.getElementById('tabSoal').classList.add('active');
                document.getElementById('viewSoal').classList.remove('hidden');
            } else {
                document.getElementById('tabKisi').classList.add('active');
                document.getElementById('viewKisi').classList.remove('hidden');
            }
        }

        form.addEventListener('submit', async function(e) {
            e.preventDefault();
            
            // Ambil semua parameter
            const mapel = document.getElementById('mapel').value;
            const kelas = document.getElementById('kelas').value;
            const topik = document.getElementById('topik').value;
            const level = document.getElementById('level').value;
            const jml_pg = document.getElementById('jml_pg').value;
            const jml_esai = document.getElementById('jml_esai').value;
            
            // Atur UI Loading
            document.getElementById('emptyState').classList.add('hidden');
            document.getElementById('resultContainer').classList.add('hidden');
            document.getElementById('errorState').classList.add('hidden');
            document.getElementById('outputToolbar').style.display = 'none';
            document.getElementById('loadingState').classList.remove('hidden');
            document.getElementById('loadingState').classList.add('flex');
            
            const btn = document.getElementById('btnSubmit');
            btn.disabled = true; btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> Memproses dengan AI...';

            try {
                // 1. Merangkai Perintah (Prompt) untuk AI
                const promptText = `Sebagai ahli pembuat soal kurikulum merdeka, tolong buatkan kisi-kisi beserta soal evaluasi berdasarkan parameter berikut:
                - Mata Pelajaran: ${mapel}
                - Kelas: ${kelas}
                - Materi Pokok / Capaian Pembelajaran: ${topik}
                - Jumlah Soal Pilihan Ganda: ${jml_pg}
                - Jumlah Soal Esai: ${jml_esai}
                - Target Level Kognitif: ${level}.
                
                Jika ada file yang dilampirkan, prioritaskan mengambil referensi bahan soal dari teks di dalam file tersebut. 
                Jadikan output menjadi format JSON terstruktur persis sesuai schema yang diminta.`;

                const partsArray = [{ text: promptText }];
                
                // Masukkan file PDF jika ada
                if (pdfBase64Data) {
                    partsArray.push({
                        inlineData: {
                            mimeType: pdfMimeType || "application/pdf",
                            data: pdfBase64Data
                        }
                    });
                }

                // 2. Skema JSON Wajib untuk Output yang Rapi
                const payload = {
                    contents: [{ parts: partsArray }],
                    systemInstruction: { parts: [{ text: "Kamu adalah asisten guru. Balas hanya dengan format JSON murni. Jangan tambahkan penjelasan pembuka atau penutup." }] },
                    generationConfig: {
                        responseMimeType: "application/json",
                        responseSchema: {
                            type: "OBJECT",
                            properties: {
                                kisi_kisi: {
                                    type: "ARRAY",
                                    items: {
                                        type: "OBJECT",
                                        properties: {
                                            nomor: { type: "INTEGER" },
                                            indikator: { type: "STRING" },
                                            level: { type: "STRING" },
                                            bentuk: { type: "STRING" },
                                            kunci: { type: "STRING" }
                                        }
                                    }
                                },
                                soal: {
                                    type: "ARRAY",
                                    items: {
                                        type: "OBJECT",
                                        properties: {
                                            tipe: { type: "STRING", description: "Wajib berisi 'pg' atau 'esai'" },
                                            nomor: { type: "INTEGER" },
                                            pertanyaan: { type: "STRING" },
                                            opsi: { type: "ARRAY", items: { type: "STRING" }, description: "Jika 'esai' kosongkan array. Jika 'pg' beri 4 opsi misal ['A. ...', 'B. ...']" },
                                            jawaban_benar: { type: "STRING" },
                                            pembahasan: { type: "STRING" }
                                        }
                                    }
                                }
                            }
                        }
                    }
                };

                // 3. Menghubungi Server Google Gemini API Asli
                // PENTING: API Key sudah dimasukkan sesuai permintaan
                const apiKey = "AIzaSyC_snrr6t7vxHt46odsGDHTEXTIQ9WSEoQ"; 
                const model = "gemini-2.5-flash-preview-09-2025";
                const url = `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`;

                let apiResult = null;
                const retries = [1000, 2000, 4000, 8000, 16000]; // Sistem Anti-Gagal (Auto Retry)
                
                for (let i = 0; i <= retries.length; i++) {
                    try {
                        const response = await fetch(url, {
                            method: 'POST',
                            headers: { 'Content-Type': 'application/json' },
                            body: JSON.stringify(payload)
                        });
                        if (!response.ok) throw new Error(`HTTP Error ${response.status}`);
                        apiResult = await response.json();
                        break; 
                    } catch (err) {
                        if (i === retries.length) throw err;
                        await new Promise(r => setTimeout(r, retries[i]));
                    }
                }

                // 4. Menerjemahkan Hasil dan Menampilkannya
                let responseText = apiResult.candidates[0].content.parts[0].text;
                
                // Mencegah error jika AI menyelipkan tag markdown
                responseText = responseText.replace(/```json/gi, '').replace(/```/gi, '').trim();
                
                currentData = JSON.parse(responseText);

                // Set Kop Ujian
                document.getElementById('lblMapel').innerText = mapel;
                document.getElementById('lblKelas').innerText = kelas;
                document.getElementById('lblMapel2').innerText = mapel;
                document.getElementById('lblKelas2').innerText = kelas;

                // Render HTML
                renderAll(currentData, mapel, kelas, topik);

                document.getElementById('loadingState').classList.add('hidden');
                document.getElementById('resultContainer').classList.remove('hidden');
                document.getElementById('outputToolbar').style.display = 'flex';
                switchTab('soal'); // Default tab ke lembar soal

            } catch (error) {
                console.error("Terjadi Kesalahan:", error);
                document.getElementById('loadingState').classList.add('hidden');
                document.getElementById('errorState').classList.remove('hidden');
                document.getElementById('errorState').classList.add('flex');
                
                // Tampilkan pesan error yang lebih detail
                let pesanError = "Gagal memproses AI. Mohon coba lagi.";
                if (error.message.includes('400')) pesanError = "API Key kosong atau salah. Jika memakai XAMPP, pastikan API Key diisi di dalam kode.";
                else if (error.message.includes('JSON')) pesanError = "Format balasan AI tidak sesuai. Silakan klik Generate lagi.";
                
                document.getElementById('errorMessage').innerText = pesanError;
            } finally {
                // Kembalikan tombol seperti semula
                btn.disabled = false; btn.innerHTML = '<i class="fa-solid fa-wand-magic-sparkles"></i> Generate Soal & Kisi-kisi';
            }
        });

        // FUNGSI RENDER (HTML & Data Persiapan Word)
        function renderAll(data, mapel, kelas, topik) {
            const soalList = document.getElementById('soalList');
            const kisiBody = document.getElementById('kisiTableBody');
            const bahasList = document.getElementById('pembahasanList');
            
            soalList.innerHTML = '';
            kisiBody.innerHTML = '';
            bahasList.innerHTML = '';

            // HTML Mentah Untuk Mode Download MS. Word
            let wordHTML = `
                <div style="font-family: 'Times New Roman', Times, serif;">
                    <h2 style="text-align:center;">KISI-KISI SOAL EVALUASI</h2>
                    <p>Mata Pelajaran : ${mapel}<br>Kelas : ${kelas}<br>Materi : ${topik}</p>
                    <table border="1" cellpadding="5" cellspacing="0" style="width:100%; border-collapse:collapse; margin-bottom: 30px;">
                        <tr style="background-color:#f2f2f2;">
                            <th>No</th><th>Indikator Soal</th><th>Level</th><th>Bentuk</th><th>Kunci</th>
                        </tr>
            `;

            // 1. Merangkai Tabel Kisi-kisi
            if(data.kisi_kisi) {
                data.kisi_kisi.forEach((k) => {
                    kisiBody.innerHTML += `
                        <tr>
                            <td class="px-4 py-3 whitespace-nowrap">${k.nomor}</td>
                            <td class="px-4 py-3 text-slate-800">${k.indikator}</td>
                            <td class="px-4 py-3 text-slate-600 text-xs font-semibold"><span class="bg-indigo-100 text-indigo-700 px-2 py-1 rounded">${k.level}</span></td>
                            <td class="px-4 py-3 text-slate-600 capitalize">${k.bentuk}</td>
                            <td class="px-4 py-3 font-bold text-primary">${k.kunci}</td>
                        </tr>
                    `;
                    wordHTML += `<tr><td align="center">${k.nomor}</td><td>${k.indikator}</td><td align="center">${k.level}</td><td align="center">${k.bentuk}</td><td align="center">${k.kunci}</td></tr>`;
                });
            }
            wordHTML += `</table><br clear="all" style="page-break-before:always" />`; 
            wordHTML += `<h2 style="text-align:center;">LEMBAR SOAL</h2>`;

            // 2. Merangkai Lembar Soal & Pembahasan
            if(data.soal) {
                data.soal.forEach((s) => {
                    const isPG = s.tipe.toLowerCase() === 'pg';
                    let htmlOpsi = '';
                    let wordOpsi = '';

                    if(isPG && s.opsi && s.opsi.length > 0) {
                        htmlOpsi = '<div class="mt-3 space-y-2">';
                        s.opsi.forEach(o => {
                            htmlOpsi += `<div class="flex gap-2 text-slate-700"><span>${o}</span></div>`;
                            wordOpsi += `<div style="margin-left: 20px;">${o}</div>`;
                        });
                        htmlOpsi += '</div>';
                    } else {
                        htmlOpsi = `<div class="mt-6 mb-2 border border-dashed border-slate-300 h-24 rounded"></div>`;
                        wordOpsi = `<br><br><br>`;
                    }

                    // Tampilan Web Soal
                    soalList.innerHTML += `
                        <div class="mb-5 pb-5 border-b border-slate-100 last:border-0 last:pb-0">
                            <div class="flex gap-3 text-slate-800">
                                <span class="font-bold">${s.nomor}.</span>
                                <div class="flex-1">
                                    <p class="font-medium text-[15px] leading-relaxed">${s.pertanyaan}</p>
                                    ${htmlOpsi}
                                </div>
                            </div>
                        </div>
                    `;

                    // Tampilan Word Soal
                    wordHTML += `
                        <div style="margin-bottom: 15px;">
                            <table width="100%" border="0" cellspacing="0" cellpadding="0">
                                <tr><td width="25" valign="top"><b>${s.nomor}.</b></td><td valign="top">${s.pertanyaan}</td></tr>
                            </table>
                            ${wordOpsi}
                        </div>
                    `;

                    // Pembahasan Web
                    bahasList.innerHTML += `
                        <div class="bg-white p-4 border border-slate-200 rounded text-sm">
                            <p class="font-bold mb-1">Soal No. ${s.nomor} <span class="text-primary ml-2">Jawaban: ${s.jawaban_benar}</span></p>
                            <p class="text-slate-600"><b>Pembahasan:</b> ${s.pembahasan}</p>
                        </div>
                    `;
                });
            }

            // Simpan HTML murni ke container tersembunyi
            wordHTML += `</div>`;
            document.getElementById('wordExportData').innerHTML = wordHTML;
        }

        function downloadWord() {
            if(!currentData) return;
            
            const mapel = document.getElementById('mapel').value.replace(/ /g, "_");
            const kelas = document.getElementById('kelas').value.replace(/ /g, "_");
            const filename = `Evaluasi_${mapel}_${kelas}.doc`;

            const htmlContent = document.getElementById('wordExportData').innerHTML;
            const header = "<html xmlns:o='urn:schemas-microsoft-com:office:office' xmlns:w='urn:schemas-microsoft-com:office:word' xmlns='http://www.w3.org/TR/REC-html40'><head><meta charset='utf-8'><title>Export HTML to Word</title></head><body>";
            const footer = "</body></html>";
            const sourceHTML = header + htmlContent + footer;
            
            const blob = new Blob(['\ufeff', sourceHTML], { type: 'application/msword' });
            const url = URL.createObjectURL(blob);
            
            const link = document.createElement('a');
            link.href = url;
            link.download = filename;
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }
